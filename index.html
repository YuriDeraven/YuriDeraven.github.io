smart-greeting.php 
<?php
/**
 * Smart Greeting System
 * Demonstrates time-zone logic and business hour validation.
 */
function get_smart_greeting($name = "Visitor") {
    date_default_timezone_set('Africa/Lagos');
    $hour = date('H');
    $day = date('l');

    // 1. Determine Greeting
    if ($hour < 12) { $greeting = "Good Morning"; }
    elseif ($hour < 17) { $greeting = "Good Afternoon"; }
    else { $greeting = "Good Evening"; }

    // 2. Business Hours Logic (Mon-Fri, 9am - 5pm)
    $is_weekend = ($day == 'Saturday' || $day == 'Sunday');
    $is_business_hours = ($hour >= 9 && $hour < 17);

    $is_open = (!$is_weekend && $is_business_hours);

    return [
        'message' => "{$greeting}, {$name}!",
        'status' => $is_open ? "Open for new projects" : "Away - Back soon!"
    ];
}
?>

// Pricing rates per project type (NGN)
const rates = {
  landing:   { min: 150000, max: 250000, days: [5, 7] },
  wordpress: { min: 300000, max: 500000, days: [10, 14] },
  webapp:    { min: 600000, max: 1000000, days: [20, 30] },
  ecommerce: { min: 500000, max: 800000, days: [15, 21] },
  api:       { min: 400000, max: 650000, days: [12, 18] },
};

function calculateQuote(type, complexity, rush) {
  const base = rates[type];
  const factor = 0.8 + complexity * 0.1;
  const rushFactor = rush ? 1.25 : 1;
  const rushDays = rush ? 0.7 : 1;

  return {
    min: Math.round(base.min * factor * rushFactor),
    max: Math.round(base.max * factor * rushFactor),
    days: base.days.map(d => Math.round(d * rushDays)),
  };
}

import re

EMAIL_RE = re.compile(r"^[\w.+-]+@[\w-]+\.[a-zA-Z]{2,}$")
PHONE_RE = re.compile(r"^(\+234|0)[789][01]\d{8}$")  # NG mobile format

def validate_email(value: str) -> bool:
    return bool(EMAIL_RE.match(value))

def validate_phone(value: str) -> bool:
    return bool(PHONE_RE.match(value.replace(" ", "")))

API Rate Limiter (Token Bucket)

A 5-token bucket refilling 1 token every 2 seconds — the same pattern used to protect production APIs from abuse. Fire requests and watch it throttle.

rate-limiter.js 
class TokenBucket {
  constructor(capacity, refillMs) {
    this.capacity = capacity;
    this.tokens = capacity;
    this.refillMs = refillMs;
    this.last = Date.now();
  }
  refill() {
    const elapsed = Date.now() - this.last;
    const add = Math.floor(elapsed / this.refillMs);
    if (add > 0) {
      this.tokens = Math.min(this.capacity, this.tokens + add);
      this.last = Date.now();
    }
  }
  tryRequest() {
    this.refill();
    if (this.tokens > 0) { this.tokens--; return true; }
    return false;
  }
}

// Express middleware
const bucket = new TokenBucket(5, 2000);
app.use((req, res, next) => {
  if (bucket.tryRequest()) return next();
  res.status(429).send('Too Many Requests');
});
